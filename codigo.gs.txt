const PERIOD_HEADERS = [
  "ID_REGISTRO", "PERIODO", "JERARQUIA", "NOMBRE_COMPLETO", "CEDULA",
  "CORREO", "TELEFONO", "PREMIO", "TIPO_COLECCION", "CANTIDAD_COLECCIONES",
  "AGENCIA_ENVIO", "SALDO_CACHEA", "COLECCION_RECIBIDA", "COMISION_ESTIMADA",
  "DISTRIBUIDOR_NOMBRE", "DISTRIBUIDOR_CEDULA", "GERENTE_NOMBRE", "GERENTE_CEDULA",
  "REGIONAL_NOMBRE", "REGIONAL_CEDULA", "LIDER_SUPERIOR", "CREADO_POR",
  "ULTIMA_ACTUALIZACION"
];

const ACCOUNTS_SHEET_NAME = "Cuentas_Sistema";
const SUMMARY_SHEET_NAME = "Resumen_Mensual";

function getOrCreatePeriodSheet(ss, periodo) {
  const sheetName = periodo ? String(periodo).trim() : "Registros";
  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(PERIOD_HEADERS);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, 14).setBackground("#1e1b4b").setFontColor("#ffffff").setFontWeight("bold");
    sheet.getRange(1, 15, 1, 2).setBackground("#0369a1").setFontColor("#ffffff").setFontWeight("bold");
    sheet.getRange(1, 17, 1, 2).setBackground("#3730a3").setFontColor("#ffffff").setFontWeight("bold");
    sheet.getRange(1, 19, 1, 5).setBackground("#581c87").setFontColor("#ffffff").setFontWeight("bold");
  } else if (sheet.getLastRow() === 0) {
    sheet.appendRow(PERIOD_HEADERS);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getOrCreateAccountsSheet(ss) {
  let sheet = ss.getSheetByName(ACCOUNTS_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(ACCOUNTS_SHEET_NAME);
    sheet.appendRow([
      "TIPO_CUENTA", "USUARIO", "CONTRASENA", "NOMBRE_COMPLETO",
      "CEDULA", "TELEFONO", "CORREO", "AGENCIA_ENVIO",
      "PM_BANCO", "PM_TELEFONO", "PM_CEDULA", "JSON_COMPLETO"
    ]);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, 12).setBackground("#0f172a").setFontColor("#ffffff").setFontWeight("bold");
  }
  return sheet;
}

function getOrCreateSummarySheet(ss) {
  let sheet = ss.getSheetByName(SUMMARY_SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SUMMARY_SHEET_NAME);
    sheet.appendRow([
      "PERIODO", "TOTAL_COLECCIONES_MES", "TOTAL_COMISIONES_USD",
      "TOTAL_SALDO_CACHEA_USD", "CANT_REGIONALES", "CANT_GERENTES",
      "CANT_DISTRIBUIDORES", "CANT_EMPRENDEDORES", "COLECCIONES_ENTREGADAS",
      "ULTIMA_SINCRONIZACION"
    ]);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, 10).setBackground("#065f46").setFontColor("#ffffff").setFontWeight("bold");
  }
  return sheet;
}

function updateMonthlySummary(ss, periodo) {
  const periodSheet = ss.getSheetByName(periodo);
  if (!periodSheet) return;
  const summarySheet = getOrCreateSummarySheet(ss);
  let totalColecciones = 0, totalComisiones = 0, totalCachea = 0;
  let countReg = 0, countGer = 0, countDist = 0, countEmp = 0, entregadas = 0;

  if (periodSheet.getLastRow() > 1) {
    const dataRange = periodSheet.getDataRange().getValues();
    const headers = dataRange.shift();
    const colIdx = name => headers.indexOf(name);
    dataRange.forEach(row => {
      const jer = String(row[colIdx("JERARQUIA")] || "").trim();
      const cantCol = Number(row[colIdx("CANTIDAD_COLECCIONES")]) || 0;
      const cachea = Number(row[colIdx("SALDO_CACHEA")]) || 0;
      const recibida = String(row[colIdx("COLECCION_RECIBIDA")] || "").trim().toUpperCase() === "SÍ";
      const comision = Number(row[colIdx("COMISION_ESTIMADA")]) || 0;
      if (jer === "Regional") countReg++;
      else if (jer === "Gerente") countGer++;
      else if (jer === "Distribuidor") countDist++;
      else if (jer === "Emprendedor") {
        countEmp++;
        totalColecciones += cantCol;
        if (recibida) entregadas++;
      }
      totalComisiones += comision;
      totalCachea += cachea;
    });
  }

  const summaryRow = [
    periodo, totalColecciones, Number(totalComisiones.toFixed(2)),
    Number(totalCachea.toFixed(2)), countReg, countGer, countDist,
    countEmp, entregadas, new Date()
  ];

  const lastRow = summarySheet.getLastRow();
  let foundRow = -1;
  if (lastRow > 1) {
    const periods = summarySheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (let i = 0; i < periods.length; i++) {
      if (String(periods[i][0]).trim() === String(periodo).trim()) {
        foundRow = i + 2;
        break;
      }
    }
  }
  if (foundRow > 0) summarySheet.getRange(foundRow, 1, 1, summaryRow.length).setValues([summaryRow]);
  else summarySheet.appendRow(summaryRow);
}

function buildRowFromRecord(periodo, data) {
  return [
    data.id || ("rec_" + new Date().getTime()), periodo,
    data.jerarquia || "Emprendedor", data.nombreCompleto || "",
    data.cedula || "", data.correo || "", data.telefono || "",
    data.premio || "Ninguno", data.tipoColeccion || "",
    Number(data.cantidadColecciones) || 0, data.agenciaEnvio || "",
    Number(data.saldoCachea) || 0, data.coleccionRecibida ? "SÍ" : "NO",
    Number(data.comisionEstimada) || 0,
    data.distribuidorNombre || "-", data.distribuidorCedula || "-",
    data.gerenteNombre || "-", data.gerenteCedula || "-",
    data.regionalNombre || "-", data.regionalCedula || "-",
    data.liderSuperior || "", data.creadoPorUsuario || "", new Date()
  ];
}

function upsertRecordInSheet(sheet, periodo, data) {
  const rowValues = buildRowFromRecord(periodo, data);
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) {
    sheet.appendRow(rowValues);
    return;
  }
  const allValues = sheet.getDataRange().getValues();
  const headers = allValues[0];
  const idCol = headers.indexOf("ID_REGISTRO");
  const jerCol = headers.indexOf("JERARQUIA");
  const cedCol = headers.indexOf("CEDULA");
  const targetId = String(data.id || "").trim();
  const targetJer = String(data.jerarquia || "").trim().toUpperCase();
  const targetCed = String(data.cedula || "").trim().toUpperCase();

  let updatedRowIndex = -1;
  for (let i = 1; i < allValues.length; i++) {
    const rowId = idCol >= 0 ? String(allValues[i][idCol] || "").trim() : "";
    const rowJer = jerCol >= 0 ? String(allValues[i][jerCol] || "").trim().toUpperCase() : "";
    const rowCed = cedCol >= 0 ? String(allValues[i][cedCol] || "").trim().toUpperCase() : "";
    if ((targetId && rowId === targetId) || (targetJer && targetCed && rowJer === targetJer && rowCed === targetCed)) {
      updatedRowIndex = i + 1;
      break;
    }
  }
  if (updatedRowIndex > 0) sheet.getRange(updatedRowIndex, 1, 1, rowValues.length).setValues([rowValues]);
  else sheet.appendRow(rowValues);
}

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (data.action === "saveSystemAccounts") {
      const accSheet = getOrCreateAccountsSheet(ss);
      accSheet.clearContents();
      accSheet.appendRow([
        "TIPO_CUENTA", "USUARIO", "CONTRASENA", "NOMBRE_COMPLETO",
        "CEDULA", "TELEFONO", "CORREO", "AGENCIA_ENVIO",
        "PM_BANCO", "PM_TELEFONO", "PM_CEDULA", "JSON_COMPLETO"
      ]);
      accSheet.getRange(1, 1, 1, 12).setBackground("#0f172a").setFontColor("#ffffff").setFontWeight("bold");
      if (data.regionalProfile) {
        const reg = data.regionalProfile, pm = reg.pagoMovil || {};
        accSheet.appendRow([
          "REGIONAL", reg.username || "regional", reg.password || "1234",
          reg.nombreCompleto || "", reg.cedula || "", reg.telefono || "",
          reg.correo || "", reg.agenciaEnvio || "",
          pm.banco || "", pm.telefono || "", pm.cedula || "", JSON.stringify(reg)
        ]);
      }
      (data.gerenteAccounts || []).forEach(g => {
        const pm = g.pagoMovil || {};
        accSheet.appendRow([
          "GERENTE", g.usuario || "", g.password || "",
          g.nombreCompleto || "", g.cedula || "", g.telefono || "",
          g.correo || "", g.agenciaEnvio || "",
          pm.banco || "", pm.telefono || "", pm.cedula || "", JSON.stringify(g)
        ]);
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "syncBulkPeriod" && data.periodo && Array.isArray(data.records)) {
      const periodo = String(data.periodo).trim();
      const sheet = getOrCreatePeriodSheet(ss, periodo);
      if (data.replaceAll) {
        sheet.clearContents();
        sheet.appendRow(PERIOD_HEADERS);
        sheet.setFrozenRows(1);
        sheet.getRange(1, 1, 1, 14).setBackground("#1e1b4b").setFontColor("#ffffff").setFontWeight("bold");
        sheet.getRange(1, 15, 1, 2).setBackground("#0369a1").setFontColor("#ffffff").setFontWeight("bold");
        sheet.getRange(1, 17, 1, 2).setBackground("#3730a3").setFontColor("#ffffff").setFontWeight("bold");
        sheet.getRange(1, 19, 1, 5).setBackground("#581c87").setFontColor("#ffffff").setFontWeight("bold");
      }
      data.records.forEach(rec => upsertRecordInSheet(sheet, periodo, rec));
      updateMonthlySummary(ss, periodo);
      return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
    }

    if (data.action === "deleteRecord" && data.periodo) {
      const periodo = String(data.periodo).trim();
      const sheet = ss.getSheetByName(periodo);
      if (sheet && sheet.getLastRow() > 1) {
        const allValues = sheet.getDataRange().getValues();
        const headers = allValues[0];
        const idCol = headers.indexOf("ID_REGISTRO");
        const jerCol = headers.indexOf("JERARQUIA");
        const cedCol = headers.indexOf("CEDULA");
        const targetId = String(data.id || "").trim();
        const targetJer = String(data.jerarquia || "").trim().toUpperCase();
        const targetCed = String(data.cedula || "").trim().toUpperCase();
        for (let i = allValues.length - 1; i >= 1; i--) {
          const rowId = idCol >= 0 ? String(allValues[i][idCol] || "").trim() : "";
          const rowJer = jerCol >= 0 ? String(allValues[i][jerCol] || "").trim().toUpperCase() : "";
          const rowCed = cedCol >= 0 ? String(allValues[i][cedCol] || "").trim().toUpperCase() : "";
          if ((targetId && rowId === targetId) || (targetJer && targetCed && rowJer === targetJer && rowCed === targetCed)) {
            sheet.deleteRow(i + 1);
          }
        }
        updateMonthlySummary(ss, periodo);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", deleted: true })).setMimeType(ContentService.MimeType.JSON);
    }

    const periodo = data.periodo ? String(data.periodo).trim() : "Registros";
    const sheet = getOrCreatePeriodSheet(ss, periodo);
    upsertRecordInSheet(sheet, periodo, data);
    updateMonthlySummary(ss, periodo);
    return ContentService.createTextOutput(JSON.stringify({ status: "ok" })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const action = e && e.parameter && e.parameter.action ? String(e.parameter.action) : "";
    if (action === "getSystemAccounts") {
      const accSheet = ss.getSheetByName(ACCOUNTS_SHEET_NAME);
      if (!accSheet || accSheet.getLastRow() <= 1) {
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", regionalProfile: null, gerenteAccounts: [], masterLeaders: [] })).setMimeType(ContentService.MimeType.JSON);
      }
      const values = accSheet.getDataRange().getValues();
      values.shift();
      let regionalProfile = null;
      const gerenteAccounts = [];
      values.forEach(row => {
        const tipo = String(row[0] || ""), rawJson = row[11];
        if (!rawJson) return;
        try {
          const parsed = JSON.parse(rawJson);
          if (tipo === "REGIONAL") regionalProfile = parsed;
          else if (tipo === "GERENTE") gerenteAccounts.push(parsed);
        } catch (err) {}
      });
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", regionalProfile, gerenteAccounts, masterLeaders: [] })).setMimeType(ContentService.MimeType.JSON);
    }
    const periodo = e && e.parameter && e.parameter.periodo ? String(e.parameter.periodo).trim() : "Registros";
    const sheet = ss.getSheetByName(periodo);
    if (!sheet || sheet.getLastRow() <= 1) {
      return ContentService.createTextOutput(JSON.stringify({ status: "ok", rows: [] })).setMimeType(ContentService.MimeType.JSON);
    }
    const values = sheet.getDataRange().getValues();
    const headers = values.shift();
    const rows = values.map(row => {
      const obj = {};
      headers.forEach((h, i) => {
        obj[h] = h === "COLECCION_RECIBIDA" ? (row[i] === true || String(row[i]).trim().toUpperCase() === "SÍ") : row[i];
      });
      return obj;
    });
    return ContentService.createTextOutput(JSON.stringify({ status: "ok", rows })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}
